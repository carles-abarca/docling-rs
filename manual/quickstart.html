Quick Start Guide

Get started with docling-rs in under 5 minutes. This guide covers the most common use cases for both the CLI and Rust library.

CLI Quick Start

1. Convert a Single File

Convert any supported document to Markdown:

Terminal
# Convert PDF to Markdown
docling-rs report.pdf --to markdown --output-dir ./output

# Convert DOCX to JSON
docling-rs document.docx --to json --output-dir ./output

# Convert to plain text
docling-rs spreadsheet.xlsx --to text

2. Batch Convert a Directory

Process multiple files at once:

Terminal
# Convert all documents in a folder
docling-rs ./documents/ --to json --output-dir ./converted

# Filter by input format
docling-rs ./documents/ --from pdf,docx --to markdown

# Continue on errors
docling-rs ./documents/ --to json --continue-on-error

3. Chunking for RAG Applications

Split documents into chunks for embedding:

Terminal
# Enable chunking with default size (512)
docling-rs document.pdf --chunk --to json

# Custom chunk size
docling-rs document.pdf --chunk --chunk-size 1000 --to markdown

# Chunk with overlap
docling-rs document.pdf --chunk --chunk-size 500 --chunk-overlap 50

Rust Library Quick Start

1. Basic Document Conversion

Rust
use docling_rs::DocumentConverter;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    // Create a converter with all enabled backends
    let converter = DocumentConverter::new();

    // Convert a file
    let result = converter.convert_file("document.pdf")?;

    // Access the converted document
    let doc = result.document();

    // Print document info
    println!("Name: {}", doc.name());
    println!("Nodes: {}", doc.nodes().len());

    Ok(())
}

2. Export to Different Formats

Rust
use docling_rs::DocumentConverter;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let converter = DocumentConverter::new();
    let result = converter.convert_file("report.docx")?;
    let doc = result.document();

    // Export to Markdown
    let markdown = doc.to_markdown();
    std::fs::write("output.md", &markdown)?;

    // Export to JSON
    let json = serde_json::to_string_pretty(doc)?;
    std::fs::write("output.json", &json)?;

    // Export to plain text
    let text = doc.to_text();
    std::fs::write("output.txt", &text)?;

    Ok(())
}

3. Convert from Bytes

Rust
use docling_rs::{DocumentConverter, InputFormat};

fn convert_from_bytes(data: Vec<u8>) -> Result<String, Box<dyn std::error::Error>> {
    let converter = DocumentConverter::new();

    // Convert bytes with explicit format
    let result = converter.convert_bytes(
        data,
        "uploaded_file.pdf".to_string(),
        InputFormat::PDF
    )?;

    Ok(result.document().to_markdown())
}

4. Working with Document Nodes

Rust
use docling_rs::{DocumentConverter, NodeType};

fn analyze_document(path: &str) -> Result<(), Box<dyn std::error::Error>> {
    let converter = DocumentConverter::new();
    let result = converter.convert_file(path)?;
    let doc = result.document();

    // Iterate over document nodes
    for node in doc.nodes() {
        match node.item().node_type() {
            NodeType::Heading => {
                println!("Heading: {}", node.item().text_content());
            }
            NodeType::Paragraph => {
                println!("Paragraph: {} chars",
                    node.item().text_content().len());
            }
            NodeType::Table => {
                println!("Table found");
            }
            _ => {}
        }
    }

    Ok(())
}

5. Chunking Documents

Rust
use docling_rs::{DocumentConverter, HierarchicalChunker};

fn chunk_document(path: &str) -> Result<Vec<String>, Box<dyn std::error::Error>> {
    let converter = DocumentConverter::new();
    let result = converter.convert_file(path)?;
    let doc = result.document();

    // Create a hierarchical chunker
    let chunker = HierarchicalChunker::new()
        .with_max_chunk_size(512)
        .with_overlap(50);

    // Chunk the document
    let chunks = chunker.chunk(doc)?;

    // Extract chunk texts
    let texts: Vec<String> = chunks
        .iter()
        .map(|c| c.text().to_string())
        .collect();

    Ok(texts)
}

Common Patterns

Error Handling

Rust
use docling_rs::{DocumentConverter, ConversionError};

fn safe_convert(path: &str) -> Result<String, String> {
    let converter = DocumentConverter::new();

    match converter.convert_file(path) {
        Ok(result) => Ok(result.document().to_markdown()),
        Err(ConversionError::FileNotFound(_)) => {
            Err("File not found".to_string())
        }
        Err(ConversionError::UnsupportedFormat(fmt)) => {
            Err(format!("Unsupported format: {}", fmt))
        }
        Err(e) => Err(format!("Conversion error: {}", e))
    }
}

Batch Processing

Rust
use docling_rs::DocumentConverter;
use std::path::Path;

fn batch_convert(dir: &Path) -> Vec<Result<String, String>> {
    let converter = DocumentConverter::new();
    let mut results = Vec::new();

    for entry in std::fs::read_dir(dir).unwrap() {
        let path = entry.unwrap().path();

        if path.is_file() {
            match converter.convert_file(&path) {
                Ok(result) => {
                    results.push(Ok(result.document().to_text()));
                }
                Err(e) => {
                    results.push(Err(format!("{}: {}",
                        path.display(), e)));
                }
            }
        }
    }

    results
}
